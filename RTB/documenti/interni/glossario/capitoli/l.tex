\letteraglossario{L}

\termine{LanguageTool}{Strumento open source di controllo ortografico, grammaticale e stilistico che supporta l'italiano; il gruppo lo usa per l'analisi automatica dei documenti.}

\termine{Latenza}{Tempo che intercorre tra una richiesta e l'inizio o il completamento della relativa risposta.}

\termine{LLM}{\emph{Large Language Model}: modello linguistico di grandi dimensioni,
  addestrato su grandi quantità di testo, in grado di interpretare e
  generare linguaggio naturale.}
