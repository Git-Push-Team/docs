\letteraglossario{O}

\termine{OAuth}{Protocollo standard di autorizzazione con cui un'applicazione ottiene un token per accedere alle \glo{API} di un altro sistema senza conoscere le credenziali dell'utente.}

\termine{OCR}{\emph{Optical Character Recognition}: tecnica che riconosce il testo
  contenuto in immagini, fotografie o documenti scansionati e lo
  converte in testo elaborabile.}
